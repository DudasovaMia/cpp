#include <iostream>
using namespace std;

int main()
{
    char meno[] = "Mia Dudasova";
    char pole2[30][30];

    // dlzka mena, -1 lebo sizeof zapocita aj znak '\0' na konci
    int dlzka = sizeof(meno) / sizeof(meno[0]) - 1;

    // cele pole vyplnim hviezdickami
    for (int i = 0; i < 30; i++)
    {
        for (int j = 0; j < 30; j++)
        {
            pole2[i][j] = '*';
        }
    }

    for (int i = 0; i < 30; i++)
    {
        // v riadku i zacina meno na pozicii i
        for (int j = 0; j < dlzka; j++)
        {
            int pozicia = i + j;

            // ked meno presiahne koniec riadku, pokracuje od zaciatku
            if (pozicia >= 30)
            {
                pozicia = pozicia - 30;
            }

            pole2[i][pozicia] = meno[j];
        }

        // posledny znak mena dam dopredu, ostatne posuniem o 1 doprava
        char posledny = meno[dlzka - 1];
        for (int k = dlzka - 1; k > 0; k--)
        {
            meno[k] = meno[k - 1];
        }
        meno[0] = posledny;
    }

    // vypis pola po riadkoch
    for (int i = 0; i < 30; i++)
    {
        for (int j = 0; j < 30; j++)
        {
            cout << pole2[i][j];
        }
        cout << "\n";
    }

    return 0;
}
